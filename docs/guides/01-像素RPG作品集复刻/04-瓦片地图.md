# 第 4 章：瓦片地图 — 加载和渲染游戏世界

> **本章目标**：在 Phaser 中加载 peteroravec.com 的像素地图，让玩家看到游戏世界。
>
> **预计时间**：3-4 小时

---

## 4.1 什么是瓦片地图？

想象拼图游戏：

```
瓦片集 (Tileset) = 一盒积木块
  ├── 🟩 草地瓦片
  ├── 🟫 道路瓦片
  ├── 🏠 建筑瓦片
  └── 🌊 水面瓦片

地图数据 (Map JSON) = 拼图说明书
  "第3行第5列放草地，第6列放道路..."

瓦片地图 = 拼完的结果
  🟩🟩🟩🟫🟫🏠
  🟩🟩🟩🟫🟫🏠
  🌊🌊🟩🟫🟫🟩
```

我们有：
- **瓦片集**：`exterior.png`（176列 × ~394行 = 69344 个 16×16 瓦片）
- **地图数据**：`final_map.json`（140×140 瓦片，24 个图层）

---

## 4.2 瓦片集兼容性问题

原站的 `exterior.png` 理论上是一个巨大的 PNG（2816×6304 像素，约 18MB），但我们下载到的只有 3.4KB——明显不完整。

本章提供两种方案：

| 方案 | 效果 | 适用场景 |
|------|------|---------|
| **A：程序化地图**（推荐先跑通） | 用不同颜色方块代替瓦片，布局完全正确 | 快速验证流程 |
| **B：真实瓦片集** | 使用完整 PNG 渲染，和原站一模一样 | 拿到完整素材后 |

---

## 4.3 方案 A：程序化生成地图

修改 `assets/js/game.js` 的 `create()` 方法：

```javascript
create() {
  console.log('🟢 create() 开始...');
  this.cameras.main.setBackgroundColor('#000000');

  // 获取地图元数据
  const mapData = this.cache.json.get('map-data');

  // 地图尺寸（瓦片单位）
  const TILE_W = 16;     // 每个瓦片 16 像素
  const TILE_H = 16;
  const MAP_W = 140;     // 140 个瓦片宽
  const MAP_H = 140;     // 140 个瓦片高
  const PIXEL_W = MAP_W * TILE_W;  // = 2240 像素
  const PIXEL_H = MAP_H * TILE_H;  // = 2240 像素

  // ---- 创建地图图像 ----
  if (mapData) {
    this.createProceduralMap(mapData, TILE_W, TILE_H, MAP_W, MAP_H);
  } else {
    // 降级：画一个简单的占位地图
    this.createPlaceholderMap(PIXEL_W, PIXEL_H);
  }

  // 保存地图信息（后续模块会用到）
  this.mapInfo = {
    width: MAP_W, height: MAP_H,
    tileWidth: TILE_W, tileHeight: TILE_H,
    pixelWidth: PIXEL_W, pixelHeight: PIXEL_H
  };

  // 设置世界边界（相机不能移出这个范围）
  this.cameras.main.setBounds(0, 0, PIXEL_W, PIXEL_H);

  console.log('✅ 地图创建完成');
}

// ---- 程序化地图 ----
createProceduralMap(mapData, tw, th, mw, mh) {
  const g = this.add.graphics();
  let tileCount = 0;

  // 瓦片 ID → 颜色映射（粗略，根据原站瓦片集估算）
  function getColor(tileId) {
    if (tileId === 0) return null;        // 0 = 空瓦片（透明）
    if (tileId < 1000)  return 0x3cb043;  // 草地绿
    if (tileId < 3000)  return 0x8b8682;  // 道路灰
    if (tileId < 5000)  return 0x4a90d9;  // 水面蓝
    if (tileId < 10000) return 0x8b7355;  // 建筑棕
    if (tileId < 30000) return 0x2d5a27;  // 树木深绿
    return 0x666666;                       // 其他
  }

  // 只渲染前 10 层（减少绘制量，大地图足够）
  const maxLayers = Math.min(10, mapData.layers.length);

  for (let li = 0; li < maxLayers; li++) {
    const layer = mapData.layers[li];
    if (layer.type !== 'tilelayer' || !layer.data) continue;

    for (let i = 0; i < layer.data.length; i++) {
      const tileId = layer.data[i];
      if (tileId === 0) continue;

      const color = getColor(tileId);
      if (color === null) continue;

      const col = i % mw;
      const row = Math.floor(i / mw);

      g.fillStyle(color, 1);
      g.fillRect(col * tw, row * th, tw, th);
      tileCount++;
    }
  }

  console.log(`🗺️ 程序化地图: 渲染了 ${tileCount} 个瓦片`);
}

// ---- 占位地图（没有 JSON 数据时使用） ----
createPlaceholderMap(pw, ph) {
  const g = this.add.graphics();
  g.fillStyle(0x3cb043, 1);  // 绿色背景
  g.fillRect(0, 0, pw, ph);
  g.fillStyle(0x8b8682, 1);  // 灰色道路
  g.fillRect(0, ph/2 - 50, pw, 100);     // 横路
  g.fillRect(pw/2 - 50, 0, 100, ph);     // 竖路
  g.fillStyle(0x8b7355, 1);  // 棕色建筑
  g.fillRect(400, 300, 120, 120);        // 建筑1
  g.fillRect(1200, 500, 120, 120);       // 建筑2
  g.fillRect(600, 1400, 160, 160);       // 建筑3
  g.fillStyle(0x4a90d9, 1);  // 蓝色水面
  g.fillRect(0, ph - 200, pw, 200);
}
```

## 4.4 方案 B：真实瓦片集（拿到完整素材后）

create() {
  // ... 前面的代码不变 ...

  // 从 JSON 创建瓦片地图
  const map = this.make.tilemap({ key: 'map-data' });

  // 将"瓦片集图片"和"JSON 中引用的瓦片集名"关联
  // final_map.json 的 tilesets[0].name 是 'exterior'
  const tileset = map.addTilesetImage('exterior', 'tileset');

  // 按需创建图层
  const groundLayer  = map.createLayer('layer1', tileset, 0, 0);

  // 设置碰撞
  wallLayer.setCollisionByExclusion([-1]); // 所有非空瓦片都碰撞

  this.roofLayer = map.createLayer('roof_concert', tileset, 0, 0);
}

## 4.5 ChunkManager：分块加载（大地图性能优化）

地图 2240×2240 像素，如果全部渲染会很卡。分块管理器只加载玩家周围的部分。

创建 `assets/js/chunk-manager.js`：

```javascript
/**
 * ChunkManager — 分块加载管理器
 *
 * 把大地图切成 N×N 块，只加载玩家周围 visibleRange 圈范围内的块。
 * 超出范围的自动销毁，节省 GPU 和内存。
 */

class ChunkManager {
  /**
   * @param {Phaser.Scene} scene
   * @param {object} mapData   — 地图 JSON 数据
   * @param {object} mapInfo   — {width, height, tileWidth, tileHeight}
   * @param {number} chunkSize — 每块多少瓦片（默认 20）
   * @param {number} visibleRange — 周围保留几圈（默认 2）
   */
  constructor(scene, mapData, mapInfo, chunkSize = 20, visibleRange = 2) {
    this.scene = scene;
    this.mapData = mapData;
    this.mapInfo = mapInfo;
    this.chunkSize = chunkSize;
    this.visibleRange = visibleRange;

    this.chunksX = Math.ceil(mapInfo.width / chunkSize);
    this.chunksY = Math.ceil(mapInfo.height / chunkSize);
    this.activeChunks = new Map();   // key → {graphics, col, row}
    this.lastCol = -1; this.lastRow = -1;

    console.log(`🧩 ChunkManager: ${this.chunksX}×${this.chunksY} 块, 每块 ${chunkSize}² 瓦片`);
  }

  /**
   * 颜色映射（和后端地图系统的颜色保持一致）
   */
  getColor(tileId) {
    if (tileId === 0) return null;
    if (tileId < 1000)  return 0x3cb043;
    if (tileId < 3000)  return 0x8b8682;
    if (tileId < 5000)  return 0x4a90d9;
    if (tileId < 10000) return 0x8b7355;
    if (tileId < 30000) return 0x2d5a27;
    return 0x666666;
  }

  /**
   * 每帧调用。玩家 X/Y 是像素坐标。
   */
  update(playerX, playerY) {
    const col = Math.floor(playerX / (this.chunkSize * this.mapInfo.tileWidth));
    const row = Math.floor(playerY / (this.chunkSize * this.mapInfo.tileHeight));

    // chunk 没变，不需要更新
    if (col === this.lastCol && row === this.lastRow) return;
    this.lastCol = col; this.lastRow = row;

    // 计算可见范围
    const minCol = Math.max(0, col - this.visibleRange);
    const maxCol = Math.min(this.chunksX - 1, col + this.visibleRange);
    const minRow = Math.max(0, row - this.visibleRange);
    const maxRow = Math.min(this.chunksY - 1, row + this.visibleRange);

    const visibleKeys = new Set();
    for (let c = minCol; c <= maxCol; c++)
      for (let r = minRow; r <= maxRow; r++)
        visibleKeys.add(`${c},${r}`);

    // 卸载不可见块
    for (const [key, chunk] of this.activeChunks) {
      if (!visibleKeys.has(key)) {
        chunk.graphics.destroy();
        this.activeChunks.delete(key);
      }
    }

    // 加载新的可见块
    for (const key of visibleKeys) {
      if (!this.activeChunks.has(key)) this.loadChunk(key);
    }
  }

  loadChunk(key) {
    const [col, row] = key.split(',').map(Number);
    const tw = this.mapInfo.tileWidth;
    const th = this.mapInfo.tileHeight;
    const x = col * this.chunkSize * tw;
    const y = row * this.chunkSize * th;

    const g = this.scene.add.graphics();
    g.setPosition(x, y);

    // 只渲染前 8 层
    const maxLayers = Math.min(8, this.mapData.layers.length);
    for (let li = 0; li < maxLayers; li++) {
      const layer = this.mapData.layers[li];
      if (layer.type !== 'tilelayer' || !layer.data) continue;

      for (let cr = 0; cr < this.chunkSize; cr++) {
        for (let cc = 0; cc < this.chunkSize; cc++) {
          const mapCol = col * this.chunkSize + cc;
          const mapRow = row * this.chunkSize + cr;
          if (mapCol >= this.mapInfo.width || mapRow >= this.mapInfo.height) continue;

          const tileId = layer.data[mapRow * this.mapInfo.width + mapCol];
          if (tileId === 0) continue;

          const color = this.getColor(tileId);
          if (color === null) continue;

          g.fillStyle(color, 1);
          g.fillRect(cc * tw, cr * th, tw, th);
        }
      }
    }

    this.activeChunks.set(key, { graphics: g, col, row });
  }

  destroy() {
    for (const [, chunk] of this.activeChunks) chunk.graphics.destroy();
    this.activeChunks.clear();
  }
}

window.ChunkManager = ChunkManager;
```

然后在 `game.js` 的 `create()` 中创建 ChunkManager：

```javascript
this.chunkManager = new ChunkManager(this, mapData, this.mapInfo, 20, 2);
```

在 `update()` 中更新：

```javascript
update(time, delta) {
  if (this.player && this.chunkManager) {
    this.player.update(delta);
    const pos = this.player.getPosition();
    this.chunkManager.update(pos.x, pos.y);
  }
}
```

## 4.7 常见踩坑

**Q：全是绿色/单一颜色？**
A：`getColor()` 中的瓦片 ID 范围是估算的。原站瓦片集有 69344 个瓦片，ID 分布很广。可以根据实际看到的颜色调整范围值。这不影响功能——后续拿到完整的瓦片集 PNG 后，渲染就完全正确了。

**Q：浏览器卡顿？**
A：减小参数：`new ChunkManager(this, mapData, this.mapInfo, 15, 1)`——更小的块 + 更少可见范围。

**Q：Graphics 绘制比 Tilemap 慢？**
A：是的，Graphics.fillRect 逐个绘制瓦片是临时的方案。拿到完整瓦片集后切换为 Phaser 的 Tilemap 系统（方案 B），性能会提升数倍。
