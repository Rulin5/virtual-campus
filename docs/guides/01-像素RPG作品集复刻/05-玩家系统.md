## 5.2 完整代码

### `assets/js/player.js`

```javascript
/**
 * player.js — 玩家角色
 *
 * 创建一个可移动的像素角色。
 * 输入来源：方向键 ↑↓←→ 或 WASD
 * 后续章节会添加触摸/虚拟摇杆支持。
 */

class Player {
  constructor(scene, x, y) {
    this.scene = scene;
    this.speed = 100;         // 像素/秒
    this.direction = 'down';  // up|down|left|right

    // 创建角色精灵（用 Graphics 画一个 16×16 的像素小人）
    this.sprite = this.createSprite(x, y);

    // 绑定键盘输入
    this.setupInput();

    console.log('✅ 玩家创建完毕 位置:', x, y);
  }

  // ---- 创建 16×16 像素角色 ----
  createSprite(x, y) {
    const g = this.scene.add.graphics();

    // 肤色（身体）
    g.fillStyle(0xffcc99, 1);
    g.fillRect(-8, -6, 16, 14);

    // 头发
    g.fillStyle(0x333333, 1);
    g.fillRect(-6, -8, 12, 4);

    // 眼睛
    g.fillStyle(0xffffff, 1);
    g.fillRect(-4, -2, 3, 2);
    g.fillRect(1, -2, 3, 2);
    g.fillStyle(0x000000, 1);
    g.fillRect(-3, -1, 2, 1);
    g.fillRect(2, -1, 2, 1);

    // 衣服
    g.fillStyle(0x3366cc, 1);
    g.fillRect(-6, 4, 12, 6);

    // 包装为 Container（Physics 需要 x/y 属性）
    const container = this.scene.add.container(x, y);
    container.add(g);

    // 添加物理体
    this.scene.physics.world.enable(container);
    container.body.setSize(16, 14);
    container.body.setOffset(-8, -7);

    return container;
  }

  // ---- 键盘输入 ----
  setupInput() {
    this.cursors = this.scene.input.keyboard.createCursorKeys();

    this.wasd = {
      up:    this.scene.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.W),
      down:  this.scene.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.S),
      left:  this.scene.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.A),
      right: this.scene.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.D),
    };

    console.log('⌨️ 键盘: 方向键 + WASD');
  }

  // ---- 每帧更新（由 game.js 的 update 调用） ----
  update(delta) {
    // 模态打开时暂停移动
    if (window.isModalOpen) {
      this.sprite.body.setVelocity(0, 0);
      return;
    }

    // 读取输入
    let vx = 0, vy = 0;
    if (this.cursors.left.isDown  || this.wasd.left.isDown)  vx = -1;
    if (this.cursors.right.isDown || this.wasd.right.isDown) vx = 1;
    if (this.cursors.up.isDown    || this.wasd.up.isDown)    vy = -1;
    if (this.cursors.down.isDown  || this.wasd.down.isDown)  vy = 1;

    // 归一对角移动速度
    if (vx !== 0 && vy !== 0) { vx *= 0.707; vy *= 0.707; }

    // 应用速度
    this.sprite.body.setVelocity(vx * this.speed, vy * this.speed);

    // 更新朝向
    if (vy < 0) this.direction = 'up';
    if (vy > 0) this.direction = 'down';
    if (vx < 0) this.direction = 'left';
    if (vx > 0) this.direction = 'right';
  }

  getPosition() {
    return { x: this.sprite.x, y: this.sprite.y };
  }
}

window.Player = Player;
```

### 修改 `assets/js/game.js`

在 `create()` 末尾添加：

```javascript
// 创建玩家（地图中央）
const cx = this.mapInfo.pixelWidth / 2;
const cy = this.mapInfo.pixelHeight / 2;
this.player = new Player(this, cx, cy);

// 相机跟随玩家（lerp 值控制平滑度：越小越平滑）
this.cameras.main.startFollow(this.player.sprite, true, 0.1, 0.1);
```

在 `update()` 中添加：

```javascript
if (this.player) {
  this.player.update(delta);
  if (this.chunkManager) {
    const pos = this.player.getPosition();
    this.chunkManager.update(pos.x, pos.y);
  }
}
```

## 5.3 验证方法

1. 刷新浏览器，点击 Play
2. 按 ↑↓←→ 或 WASD —— 角色应该在地图上移动
3. 相机应该跟随角色
4. 打开模态窗口时（按 F12 输入 `window.isModalOpen = true`），角色停止
5. 对角移动（同时按右+下）速度不应变快（因为做了归一化）

## 5.4 常见踩坑

**Q：角色动不了？**
A：1. F12 Console 看报错 2. 确认点击了画布（Phaser 需要焦点）3. 确认 `window.isModalOpen` 不是 true

**Q：角色走出地图？**
A：检查 `setBounds` 和物理碰撞是否设置。ChunkManager 只控制渲染，不控制边界。需要在 create() 中设置 `this.physics.world.setBounds(0, 0, pw, ph)` 并给 player.sprite.body 设置 `setCollideWorldBounds(true)`。
