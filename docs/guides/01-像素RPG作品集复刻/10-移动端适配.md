## 1. 核心代码：`assets/js/virtual-joystick.js`

> **设计说明：** 虚拟摇杆由两个圆组成：固定的外圆（底座）和跟随手指的内圆（摇杆头）。使用 HTML5 Canvas API 绘制在独立的小 canvas 上，通过绝对定位放在屏幕左下角。

```javascript
/**
 * VirtualJoystick — 移动端虚拟摇杆
 *
 * 工作原理：
 * 1. 检测设备是否支持触摸
 * 2. 在屏幕左下角创建一个隐藏的小 canvas
 * 3. 触摸开始时显示摇杆，手指移动时更新内圆位置
 * 4. 输出归一化方向向量 { dx: -1~1, dy: -1~1 }
 *
 * 用法：
 *   const joystick = new VirtualJoystick();
 *   // 在 update 中读取：
 *   const dir = joystick.getDirection();
 *   // dir.dx 和 dir.dy 取值范围为 -1 到 1
 */
class VirtualJoystick {

  /**
   * @param {object} options — 可选配置
   * @param {number} options.outerRadius — 外圆半径，默认 60
   * @param {number} options.innerRadius — 内圆半径，默认 25
   * @param {string} options.position — 位置：'bottom-left' | 'bottom-right'，默认 'bottom-left'
   * @param {number} options.margin — 距屏幕边缘的间距，默认 30
   */
  constructor(options = {}) {
    this.outerRadius = options.outerRadius || 60;
    this.innerRadius = options.innerRadius || 25;
    this.position = options.position || 'bottom-left';
    this.margin = options.margin || 30;

    // 摇杆状态
    this.active = false;
    this.touchId = null;       // 跟踪当前触摸点的 ID
    this.baseX = 0;            // 外圆中心 X（屏幕坐标）
    this.baseY = 0;            // 外圆中心 Y（屏幕坐标）
    this.stickX = 0;           // 内圆相对位移 X
    this.stickY = 0;           // 内圆相对位移 Y

    // 输出方向向量（-1 ~ 1）
    this.dx = 0;
    this.dy = 0;

    // ===== 检测触摸支持 =====
    this.isTouchDevice = ('ontouchstart' in window) ||
                         (navigator.maxTouchPoints > 0) ||
                         (navigator.msMaxTouchPoints > 0);

    // 仅在触摸设备上创建摇杆
    if (!this.isTouchDevice) {
      console.log('[Joystick] Not a touch device — joystick disabled');
      return;
    }

    console.log('[Joystick] Touch device detected — creating joystick');

    // ===== 创建 Canvas =====
    this.canvas = document.createElement('canvas');
    this.canvas.id = 'virtual-joystick';
    this.canvas.width = this.outerRadius * 2 + 10;
    this.canvas.height = this.outerRadius * 2 + 10;

    // 使用 CSS 定位
    this.canvas.style.cssText = `
      position: fixed;
      ${this.position.includes('bottom') ? 'bottom' : 'top'}: ${this.margin}px;
      ${this.position.includes('left') ? 'left' : 'right'}: ${this.margin}px;
      z-index: 999;
      opacity: 0.5;
      pointer-events: auto;
      touch-action: none;
    `;
    document.body.appendChild(this.canvas);

    this.ctx = this.canvas.getContext('2d');

    // 外圆中心（在 canvas 坐标系中）
    this.baseX = this.outerRadius + 5;
    this.baseY = this.outerRadius + 5;

    // 内圆初始位置（居中）
    this.stickX = 0;
    this.stickY = 0;

    // ===== 绑定事件 =====
    this._bindEvents();

    // 初始绘制
    this._draw();

    console.log('[Joystick] Canvas created:', this.canvas.width, 'x', this.canvas.height);
  }

  // ===================================================================
  //  公开方法
  // ===================================================================

  /**
   * 获取当前的归一化方向向量
   * @returns {{ dx: number, dy: number }}
   */
  getDirection() {
    return { dx: this.dx, dy: this.dy };
  }

  /**
   * 摇杆是否正在被使用
   * @returns {boolean}
   */
  isActive() {
    return this.active;
  }

  // ===================================================================
  //  内部方法
  // ===================================================================

  /**
   * 绑定触摸事件
   */
  _bindEvents() {
    // 使用 addEventListener 绑定到 document
    // 这样即使手指滑出 canvas 也能跟踪

    this._onTouchStart = this._handleTouchStart.bind(this);
    this._onTouchMove  = this._handleTouchMove.bind(this);
    this._onTouchEnd   = this._handleTouchEnd.bind(this);

    document.addEventListener('touchstart', this._onTouchStart, { passive: false });
    document.addEventListener('touchmove',  this._onTouchMove,  { passive: false });
    document.addEventListener('touchend',   this._onTouchEnd);
    document.addEventListener('touchcancel',this._onTouchEnd);
  }

  /**
   * 触摸开始：如果手指落在左下角区域，激活摇杆
   */
  _handleTouchStart(e) {
    e.preventDefault();

    for (let i = 0; i < e.changedTouches.length; i++) {
      const touch = e.changedTouches[i];
      const cx = touch.clientX;
      const cy = touch.clientY;

      // 判断触摸点是否在摇杆允许区域内（屏幕左下角 150x150 区域）
      const touchZoneSize = this.outerRadius * 3;
      const isInZone = (
        cy > window.innerHeight - touchZoneSize &&
        cx < touchZoneSize
      );

      if (isInZone && !this.active) {
        this.active = true;
        this.touchId = touch.identifier;

        // 不需要移动外圆位置，保持固定在左下角
        // 只更新内圆位移
        this._updateStickPosition(cx, cy);

        this.canvas.style.opacity = '0.8';
        this._draw();
        break;
      }
    }
  }

  /**
   * 触摸移动：更新内圆位置
   */
  _handleTouchMove(e) {
    e.preventDefault();

    if (!this.active) return;

    for (let i = 0; i < e.changedTouches.length; i++) {
      const touch = e.changedTouches[i];
      if (touch.identifier === this.touchId) {
        this._updateStickPosition(touch.clientX, touch.clientY);
        this._draw();
        break;
      }
    }
  }

  /**
   * 触摸结束：重置摇杆
   */
  _handleTouchEnd(e) {
    for (let i = 0; i < e.changedTouches.length; i++) {
      const touch = e.changedTouches[i];
      if (touch.identifier === this.touchId) {
        this.active = false;
        this.touchId = null;
        this.stickX = 0;
        this.stickY = 0;
        this.dx = 0;
        this.dy = 0;
        this.canvas.style.opacity = '0.5';
        this._draw();
        break;
      }
    }
  }

  /**
   * 根据手指位置更新内圆位移，并计算方向向量
   * @param {number} fingerX — 手指屏幕 X 坐标
   * @param {number} fingerY — 手指屏幕 Y 坐标
   */
  _updateStickPosition(fingerX, fingerY) {
    // 计算手指相对于 canvas 左上角的偏移
    const canvasRect = this.canvas.getBoundingClientRect();
    const relX = fingerX - canvasRect.left - this.baseX;
    const relY = fingerY - canvasRect.top - this.baseY;

    // 限制在外圆半径内
    const dist = Math.sqrt(relX * relX + relY * relY);
    const maxDist = this.outerRadius - this.innerRadius;

    if (dist > maxDist) {
      const ratio = maxDist / dist;
      this.stickX = relX * ratio;
      this.stickY = relY * ratio;
    } else {
      this.stickX = relX;
      this.stickY = relY;
    }

    // 输出归一化方向向量 (-1 ~ 1)
    this.dx = this.stickX / maxDist;
    this.dy = this.stickY / maxDist;

    // 舍入到 2 位小数，避免微小的浮点漂移
    this.dx = Math.round(this.dx * 100) / 100;
    this.dy = Math.round(this.dy * 100) / 100;
  }

  /**
   * 绘制摇杆
   */
  _draw() {
    const ctx = this.ctx;
    const w = this.canvas.width;
    const h = this.canvas.height;

    // 清除画布
    ctx.clearRect(0, 0, w, h);

    const bx = this.baseX;
    const by = this.baseY;
    const sx = bx + this.stickX;
    const sy = by + this.stickY;

    // —— 外圆（底座）——
    ctx.beginPath();
    ctx.arc(bx, by, this.outerRadius, 0, Math.PI * 2);
    ctx.fillStyle = 'rgba(255, 255, 255, 0.15)';
    ctx.fill();
    ctx.strokeStyle = 'rgba(255, 255, 255, 0.4)';
    ctx.lineWidth = 3;
    ctx.stroke();

    // —— 内圆（摇杆头）——
    ctx.beginPath();
    ctx.arc(sx, sy, this.innerRadius, 0, Math.PI * 2);
    ctx.fillStyle = this.active
      ? 'rgba(233, 69, 96, 0.7)'    // 活跃时：红色
      : 'rgba(255, 255, 255, 0.3)'; // 闲置时：白色半透明
    ctx.fill();
    ctx.strokeStyle = 'rgba(255, 255, 255, 0.6)';
    ctx.lineWidth = 2;
    ctx.stroke();
  }

  /**
   * 销毁摇杆（移除 DOM 元素和事件）
   */
  destroy() {
    if (this.canvas && this.canvas.parentNode) {
      this.canvas.parentNode.removeChild(this.canvas);
    }
    document.removeEventListener('touchstart', this._onTouchStart);
    document.removeEventListener('touchmove',  this._onTouchMove);
    document.removeEventListener('touchend',   this._onTouchEnd);
    document.removeEventListener('touchcancel',this._onTouchEnd);
    console.log('[Joystick] Destroyed');
  }
}
```

## 2. 修改 `assets/js/player.js` — 读取摇杆输入

在 Player 类的构造函数中接收摇杆引用，在 `update()` 中优先读取摇杆：

```javascript
// ===== player.js 修改 =====

class Player {

  /**
   * @param {Phaser.Scene} scene
   * @param {number} x — 初始 X
   * @param {number} y — 初始 Y
   * @param {VirtualJoystick} joystick — （新增）虚拟摇杆引用
   */
  constructor(scene, x, y, joystick) {
    this.scene = scene;

    // ★ 新增：保存摇杆引用
    this.joystick = joystick || null;

    // ... 原有的代码（创建 sprite、设置物理体等）...

    // ===== 键盘控制（原有的） =====
    this.cursors = scene.input.keyboard.createCursorKeys();
    this.wasd = {
      up:    scene.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.W),
      down:  scene.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.S),
      left:  scene.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.A),
      right: scene.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.D),
    };

    // 移动速度
    this.speed = 160;  // 像素/秒
  }

  update() {
    // 模态打开时禁止移动
    if (window.isModalOpen) {
      this.body.setVelocity(0, 0);
      return;
    }

    let vx = 0;
    let vy = 0;

    // ===== 优先读取摇杆输入（移动端） =====
    if (this.joystick && this.joystick.isActive()) {
      const dir = this.joystick.getDirection();
      vx = dir.dx * this.speed;
      vy = dir.dy * this.speed;

      // 更新朝向（面朝移动方向）
      if (dir.dx > 0.1) {
        this.setFlipX(false);   // 面向右
      } else if (dir.dx < -0.1) {
        this.setFlipX(true);    // 面向左
      }
    }
    // ===== 否则读取键盘输入 =====
    else {
      if (this.cursors.left.isDown  || this.wasd.left.isDown)  { vx = -this.speed; this.setFlipX(true); }
      if (this.cursors.right.isDown || this.wasd.right.isDown) { vx =  this.speed; this.setFlipX(false); }
      if (this.cursors.up.isDown    || this.wasd.up.isDown)    { vy = -this.speed; }
      if (this.cursors.down.isDown  || this.wasd.down.isDown)  { vy =  this.speed; }
    }

    // 归一化对角线速度
    if (vx !== 0 && vy !== 0) {
      const factor = 1 / Math.sqrt(2);
      vx *= factor;
      vy *= factor;
    }

    // 应用速度
    this.body.setVelocity(vx, vy);
  }
}
```

## 6. 常见坑点

| 问题 | 原因 | 解决方法 |
|------|------|----------|
| 摇杆不显示 | `isTouchDevice` 判断返回 false | 在模拟器中确保勾选了"模拟触摸事件"；或临时强制 `this.isTouchDevice = true` |
| 手指滑动时页面跟着滚动 | 未阻止默认触摸行为 | `touchstart` 和 `touchmove` 事件需要 `{ passive: false }` 选项 |
| 摇杆位置偏移 | canvas 的 `getBoundingClientRect()` 返回值与 page 偏移不匹配 | 使用 `clientX/clientY` 减去 canvas 的 `left/top`，而不使用 `pageX/pageY` |
| 多个手指同时触摸导致摇杆跳动 | 未跟踪 `touch.identifier` | 使用 `touchId` 变量锁定当前手指，忽略其他触摸点 |
| 摇杆区域太小不好按 | `touchZoneSize` 太小 | 增大触摸判定区域（当前为 `outerRadius * 3`） |
| 手机上模态太大无法关闭 | 未适配移动端 CSS | 确保 `.modal-close` 按钮足够大，且模态不超出屏幕 |
| 画布被摇杆遮挡 | z-index 层级问题 | 摇杆 z-index: 999，画布应该是默认层级 |

// 在 game.js 中临时添加：

this.joystick.isTouchDevice = true;

this.joystick.canvas.addEventListener('mousedown', (e) => {
  this.joystick.active = true;

  this.joystick._draw();

this.joystick.canvas.addEventListener('mousemove', (e) => {
  if (this.joystick.active) {
    this.joystick._updateStickPosition(e.clientX, e.clientY);
    this.joystick._draw();

this.joystick.canvas.addEventListener('mouseup', () => {
  this.joystick.active = false;
  this.joystick.stickX = 0;
  this.joystick.stickY = 0;
  this.joystick.dx = 0;
  this.joystick.dy = 0;
  this.joystick._draw();
});

## 章节小结

- **VirtualJoystick** 使用独立 Canvas 绘制，通过 CSS 固定在左下角
- **自动检测**触摸设备，桌面端不启用（零干扰）
- **方向向量归一化**后传给 Player，确保各方向移动速度一致
- **响应式 CSS** 通过媒体查询适配 480px、768px、1200px 三个断点
- 键盘和摇杆可以**无缝共存**，摇杆激活时优先
