# 第7章 · 模态窗口系统 (Modal System)

> **Phase 1** — 核心基础搭建 | **预计时间：** 45-60 分钟

---

## 本章目标

- 创建一个 `ModalManager` 类，管理 10+ 个模态窗口
- 实现统一的打开/关闭逻辑（`open()` / `close()`）
- 支持多种关闭方式：关闭按钮、点击背景、ESC 键
- 绑定"Under the Hood"标签切换、CV 项目折叠、Memo 卡片翻页
- 暴露 `window.isModalOpen` 标记，配合游戏暂停/恢复

---

## 2. 核心代码：`assets/js/modal-manager.js`

> **说明：** 以下代码完全自包含，复制到文件中即可使用。每段都有注释说明。

```javascript
/**
 * ModalManager — 管理所有模态窗口
 * 
 * 功能：
 * - 统一打开/关闭模态
 * - 多种关闭触发方式（按钮、背景、ESC）
 * - 标签页切换、CV 折叠、Memo 翻页
 * - 游戏暂停/恢复标记
 */
class ModalManager {

  /**
   * @param {Phaser.Scene} scene — 当前游戏场景
   */
  constructor(scene) {
    this.scene = scene;

    // ========== 存储所有模态的 DOM 引用 ==========
    this.modals = {
      tech:      document.getElementById('modal-tech'),
      about:     document.getElementById('modal-about'),
      cv:        document.getElementById('modal-cv'),
      projects:  document.getElementById('modal-projects'),
      contact:   document.getElementById('modal-contact'),
      memo1:     document.getElementById('modal-memo1'),
      memo2:     document.getElementById('modal-memo2'),
      memo3:     document.getElementById('modal-memo3'),
      memo4:     document.getElementById('modal-memo4'),
      memo5:     document.getElementById('modal-memo5'),
      memo6:     document.getElementById('modal-memo6'),
      bigmap:    document.getElementById('modal-bigmap'),
      underhood: document.getElementById('modal-underhood'),
    };

    this.backdrop = document.getElementById('modal-backdrop');

    // 当前打开的模态 ID（null 表示全部关闭）
    this.currentModalId = null;

    // 全局标记：供游戏主循环使用
    window.isModalOpen = false;

    // ========== 初始化所有事件绑定 ==========
    this._bindCloseButtons();
    this._bindBackdropClick();
    this._bindEscapeKey();
    this._bindScrollUpButtons();
    this._bindUnderHoodTabs();
    this._bindCvToggles();
    this._bindMemoNavigation();

    console.log('[ModalManager] Initialized with modals:', Object.keys(this.modals));
  }

  // ===================================================================
  //  公开方法
  // ===================================================================

  /**
   * 打开指定的模态窗口
   * @param {string} modalId - 模态的 key（如 'tech', 'about', 'memo1'）
   */
  open(modalId) {
    const modal = this.modals[modalId];
    if (!modal) {
      console.warn('[ModalManager] Modal not found:', modalId);
      return;
    }

    // 先关闭当前打开的（防止同时打开两个）
    if (this.currentModalId) {
      this.close();
    }

    this.currentModalId = modalId;
    modal.classList.add('active');
    this.backdrop.classList.add('active');
    window.isModalOpen = true;

    // 暂停游戏场景
    this.scene.scene.pause();
    // 如果输入插件可用，禁用键盘输入
    if (this.scene.input && this.scene.input.keyboard) {
      this.scene.input.keyboard.enabled = false;
    }

    console.log('[ModalManager] Opened:', modalId);
  }

  /**
   * 关闭当前模态窗口
   */
  close() {
    if (!this.currentModalId) return;

    const modal = this.modals[this.currentModalId];
    if (modal) {
      modal.classList.remove('active');
    }

    this.backdrop.classList.remove('active');
    window.isModalOpen = false;

    // 恢复游戏场景
    this.scene.scene.resume();
    if (this.scene.input && this.scene.input.keyboard) {
      this.scene.input.keyboard.enabled = true;
    }

    console.log('[ModalManager] Closed:', this.currentModalId);
    this.currentModalId = null;
  }

  // ===================================================================
  //  私有方法 — 事件绑定
  // ===================================================================

  /**
   * 绑定所有模态内的关闭按钮（×）
   */
  _bindCloseButtons() {
    // 找到所有带有 .modal-close 类的按钮
    const closeButtons = document.querySelectorAll('.modal-close');
    closeButtons.forEach(btn => {
      btn.addEventListener('click', (e) => {
        e.stopPropagation();
        this.close();
      });
    });
    console.log('[ModalManager] Bound', closeButtons.length, 'close buttons');
  }

  /**
   * 绑定背景遮罩点击关闭
   */
  _bindBackdropClick() {
    this.backdrop.addEventListener('click', () => {
      this.close();
    });
  }

  /**
   * 绑定 ESC 键关闭
   */
  _bindEscapeKey() {
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && window.isModalOpen) {
        this.close();
      }
    });
  }

  /**
   * 绑定"Scroll Up"按钮（某些模态内有返回顶部的按钮）
   * 选择器：.scroll-up-btn
   */
  _bindScrollUpButtons() {
    const scrollUpBtns = document.querySelectorAll('.scroll-up-btn');
    scrollUpBtns.forEach(btn => {
      btn.addEventListener('click', (e) => {
        e.stopPropagation();
        // 找到最近的模态内容区域并滚动到顶部
        const modalContent = btn.closest('.modal-content');
        if (modalContent) {
          modalContent.scrollTo({ top: 0, behavior: 'smooth' });
        }
      });
    });
    console.log('[ModalManager] Bound', scrollUpBtns.length, 'scroll-up buttons');
  }

  /**
   * 绑定 Under the Hood 标签页切换
   * 使用 data-tab 属性关联面板 id
   */
  _bindUnderHoodTabs() {
    const tabs = document.querySelectorAll('.hood-tab');
    tabs.forEach(tab => {
      tab.addEventListener('click', (e) => {
        e.stopPropagation();
        const targetPanelId = tab.getAttribute('data-tab');

        // 移除所有标签的活动状态
        tabs.forEach(t => t.classList.remove('active'));
        tab.classList.add('active');

        // 隐藏所有面板
        const panels = document.querySelectorAll('.hood-panel');
        panels.forEach(p => p.classList.remove('active'));

        // 显示目标面板
        const targetPanel = document.getElementById(targetPanelId);
        if (targetPanel) {
          targetPanel.classList.add('active');
        }
      });
    });

    // 默认激活第一个标签
    if (tabs.length > 0) {
      tabs[0].click();
    }
    console.log('[ModalManager] Bound', tabs.length, 'hood tabs');
  }

  /**
   * 绑定 CV 项目折叠/展开按钮
   * 使用 data-cv-target 属性关联详情 div 的 id
   */
  _bindCvToggles() {
    const toggles = document.querySelectorAll('.cv-toggle');
    toggles.forEach(toggle => {
      toggle.addEventListener('click', (e) => {
        e.stopPropagation();
        const targetId = toggle.getAttribute('data-cv-target');
        const detail = document.getElementById(targetId);
        if (!detail) return;

        // 切换显示/隐藏
        if (detail.style.display === 'none' || detail.style.display === '') {
          detail.style.display = 'block';
          toggle.classList.add('expanded');
        } else {
          detail.style.display = 'none';
          toggle.classList.remove('expanded');
        }
      });
    });
    console.log('[ModalManager] Bound', toggles.length, 'CV toggles');
  }

  /**
   * 绑定 Memo 卡片的前后翻页
   * 每个 memo 模态内有 .memo-prev 和 .memo-next 按钮
   * 点击后切换到前/后一个 memo 模态
   */
  _bindMemoNavigation() {
    // 所有 memo 模态的 ID 列表
    this.memoIds = ['memo1', 'memo2', 'memo3', 'memo4', 'memo5', 'memo6'];

    this.memoIds.forEach((memoId, index) => {
      const modal = this.modals[memoId];
      if (!modal) return;

      // 上一页按钮
      const prevBtn = modal.querySelector('.memo-prev');
      if (prevBtn) {
        prevBtn.addEventListener('click', (e) => {
          e.stopPropagation();
          // 循环：到第1个时跳到第6个
          const prevIdx = (index - 1 + this.memoIds.length) % this.memoIds.length;
          this.close();
          this.open(this.memoIds[prevIdx]);
        });
      }

      // 下一页按钮
      const nextBtn = modal.querySelector('.memo-next');
      if (nextBtn) {
        nextBtn.addEventListener('click', (e) => {
          e.stopPropagation();
          // 循环：到第6个时跳到第1个
          const nextIdx = (index + 1) % this.memoIds.length;
          this.close();
          this.open(this.memoIds[nextIdx]);
        });
      }
    });
    console.log('[ModalManager] Bound memo navigation for', this.memoIds.length, 'memos');
  }
}
```

## 3. 在 `game.js` 中初始化

在 `assets/js/game.js` 的 `create()` 方法中，创建 ModalManager 实例：

```javascript
// ===== game.js 的 create() 方法中新增 =====

create() {
  // ... 已有的代码（地图、玩家、NPC 等）

  // ★ 创建模态管理器
  this.modalManager = new ModalManager(this);

  // 将实例也挂到 window 上，方便其他脚本访问
  window.modalManager = this.modalManager;
}
```

## 4. 如何使用

在游戏中的任何地方打开模态窗口，只需：

```javascript
// 打开科技栈模态
window.modalManager.open('tech');

// 打开第3张 Memo 卡片
window.modalManager.open('memo3');

// 关闭当前模态（ModalManager 内部自动处理）
window.modalManager.close();
```

配合第9章的交互区域，典型用法是：

```javascript
// 在交互区域的回调中
if (Phaser.Input.Keyboard.JustDown(this.eKey)) {
  this.modalManager.open(zone.modalId);
}
```

## 6. 常见坑点

| 问题 | 原因 | 解决方法 |
|------|------|----------|
| `modal is null` 报错 | HTML 中的 id 和 JS 中 `this.modals` 的 key 不对应 | 检查 `index.html` 中每个模态的 `id` 属性是否与 `modal-manager.js` 第 21-35 行的 key 完全一致 |
| 模态打开后背景不能滚动 | 这是预期行为；如果想保留滚动，给 `.modal-window.active` 加 `overflow-y: auto` | 已在 CSS 中包含 |
| 点击模态内部也会关闭 | 事件冒泡到了 backdrop | 在模态窗口上阻止冒泡：`modal.addEventListener('click', e => e.stopPropagation())` |
| 按 ESC 关闭的是浏览器标签页而不是模态 | Phaser 可能捕获了键盘事件 | 优先用 `window.isModalOpen` 判断，或使用 `e.preventDefault()` |
| Memo 翻页后游戏无法操作 | 翻页时未正确恢复场景 | 已处理：`close()` 和 `open()` 内部自动 resume/pause |
| 多个模态重叠 | 未先关闭再打开 | `open()` 方法已先调用 `close()` 再打开新窗口 |
| Under the Hood 标签切换不生效 | data-tab 属性值与面板 id 不匹配 | 确保 `<button data-tab="hood-overview">` 对应的面板是 `<div id="hood-overview">` |
