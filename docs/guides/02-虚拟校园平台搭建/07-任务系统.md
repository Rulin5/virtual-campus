## 7.2 后端 API

```go
// handler/quest.go

// GET /api/quests — 获取校园内所有可用任务
func ListQuests(c *gin.Context) {
    campusID := c.Query("campusId")
    rows, _ := repository.DB.Query(
        `SELECT id, title, description, objectives_json, rewards_json
         FROM quests WHERE campus_id = $1 AND is_active = true`, campusID)

    var quests []Quest
    for rows.Next() {
        var q Quest
        rows.Scan(&q.ID, &q.Title, &q.Description, &q.ObjectivesJSON, &q.RewardsJSON)
        quests = append(quests, q)
    }
    c.JSON(200, gin.H{"quests": quests})
}

// POST /api/quests/:id/accept — 接受任务
func AcceptQuest(c *gin.Context) {
    userID, _ := c.Get("userID")
    questID := c.Param("id")

    // 检查是否已接受
    var exists bool
    repository.DB.QueryRow(
        `SELECT EXISTS(SELECT 1 FROM user_quests
         WHERE user_id=$1 AND quest_id=$2 AND status='in_progress')`,
        userID, questID).Scan(&exists)
    if exists { c.JSON(409, gin.H{"error":"already accepted"}); return }

    // 插入记录
    repository.DB.Exec(
        `INSERT INTO user_quests (user_id, quest_id, status, progress_json)
         VALUES ($1, $2, 'in_progress', '{}')`, userID, questID)

    c.JSON(201, gin.H{"status": "accepted"})
}

// POST /api/quests/:id/complete — 完成任务
func CompleteQuest(c *gin.Context) {
    userID, _ := c.Get("userID")
    questID := c.Param("id")

    // 更新状态
    repository.DB.Exec(
        `UPDATE user_quests SET status='completed', completed_at=NOW()
         WHERE user_id=$1 AND quest_id=$2 AND status='in_progress'`,
        userID, questID)

    // 发放奖励
    var rewardsJSON string
    repository.DB.QueryRow(
        `SELECT rewards_json FROM quests WHERE id=$1`, questID,
    ).Scan(&rewardsJSON)

    c.JSON(200, gin.H{"status": "completed", "rewards": rewardsJSON})
}
```

## 7.3 前端任务面板

```javascript
// quest-ui.js
class QuestUI {
  constructor(scene) {
    this.scene = scene;
    this.activeQuests = [];
    this.panel = null;
  }

  // 按 Q 键打开/关闭任务面板
  toggle(campusId) {
    if (this.panel) { this.hide(); return; }
    this.show(campusId);
  }

  async show(campusId) {
    const resp = await fetch(`/api/quests?campusId=${campusId}`);
    const data = await resp.json();

    const cam = this.scene.cameras.main;
    this.panel = this.scene.add.container(0, 0)
      .setScrollFactor(0).setDepth(2000);

    const bg = this.scene.add.graphics();
    bg.fillStyle(0x000000, 0.85);
    bg.fillRoundedRect(cam.width/2 - 200, 40, 400, 300, 4);
    this.panel.add(bg);

    const title = this.scene.add.text(cam.width/2, 55,
      '📋 QUESTS', {
        fontFamily: '"Press Start 2P"',
        fontSize: '10px', color: '#ffcc00'
      }).setOrigin(0.5, 0);
    this.panel.add(title);

    data.quests.forEach((q, i) => {
      const t = this.scene.add.text(cam.width/2 - 180, 90 + i * 40,
        `${q.title}\n${q.description}`, {
          fontFamily: '"Press Start 2P"',
          fontSize: '7px', color: '#fff',
          wordWrap: { width: 340 }, lineSpacing: 3
        });
      this.panel.add(t);
    });

    const hint = this.scene.add.text(cam.width/2, 320,
      'Press Q to close', {
        fontFamily: '"Press Start 2P"',
        fontSize: '6px', color: '#888'
      }).setOrigin(0.5, 0);
    this.panel.add(hint);
  }

  hide() {
    if (this.panel) { this.panel.destroy(); this.panel = null; }
  }
}
```
