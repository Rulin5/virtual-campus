### E2E 测试：完整用户闭环

`apps/web/tests/full-flow.spec.ts`：

```typescript
import { test, expect } from '@playwright/test';

test('完整闭环：注册→登录→选校园→进地图→对话→做任务', async ({ page }) => {
  // ===== Step 1: 注册 =====
  await page.goto('http://localhost/login');
  await page.click('text=Register');

  const username = `testuser_${Date.now()}`;
  await page.fill('input[placeholder="Username"]', username);
  await page.fill('input[placeholder="Email"]', `${username}@test.com`);
  await page.fill('input[placeholder="Password"]', '123456');
  await page.click('button[type="submit"]');

  // 验证：跳转到登录页
  await expect(page).toHaveURL(/login/);

  // ===== Step 2: 登录 =====
  await page.fill('input[placeholder="Email"]', `${username}@test.com`);
  await page.fill('input[placeholder="Password"]', '123456');
  await page.click('button[type="submit"]');

  // 验证：跳转到校园选择页
  await expect(page).toHaveURL(/campuses/);

  // ===== Step 3: 选择校园 =====
  const campusCard = page.locator('.nes-btn').first();
  await expect(campusCard).toBeVisible();
  await campusCard.click();

  // 验证：进入游戏页
  await expect(page).toHaveURL(/campus\//);

  // ===== Step 4: 等待游戏加载 =====
  // Phaser Canvas 应该出现
  const canvas = page.locator('canvas');
  await expect(canvas).toBeVisible({ timeout: 10000 });

  // ===== Step 5: 模拟键盘移动 =====
  await canvas.click(); // 获取焦点

  // 按右箭头移动
  await page.keyboard.press('ArrowRight');
  await page.waitForTimeout(500);
  await page.keyboard.press('ArrowDown');
  await page.waitForTimeout(500);

  // ===== Step 6: 按 E 与 NPC 交互 =====
  // 先走到 NPC 附近（在实际测试中需要知道 NPC 坐标）
  await page.keyboard.press('KeyE');

  // 对话气泡应该出现
  // await expect(page.locator('.dialogue-container')).toBeVisible();

  // ===== Step 7: 关闭对话 =====
  await page.keyboard.press('Escape');

  // ===== Step 8: 打开任务面板 =====
  await page.keyboard.press('KeyQ');
  // await expect(page.locator('.quest-panel')).toBeVisible();
  await page.keyboard.press('KeyQ'); // 关闭

  // ===== Step 9: 验证 API 调用成功 =====
  // 检查 Console 没有错误
  page.on('console', msg => {
    if (msg.type() === 'error') {
      console.error('❌ Console error:', msg.text());
    }
  });
});

test('API 接口健康检查', async ({ request }) => {
  // 健康检查
  const health = await request.get('http://localhost:8080/api/health');
  expect(health.status()).toBe(200);
  const healthData = await health.json();
  expect(healthData.status).toBe('ok');

  // 注册
  const register = await request.post('http://localhost:8080/api/auth/register', {
    data: {
      username: 'apitest',
      email: 'apitest@test.com',
      password: '123456'
    }
  });
  expect([201, 409]).toContain(register.status()); // 409 = 已注册

  // 登录
  const login = await request.post('http://localhost:8080/api/auth/login', {
    data: { email: 'apitest@test.com', password: '123456' }
  });
  expect(login.status()).toBe(200);
  const loginData = await login.json();
  expect(loginData.token).toBeDefined();

  // 获取个人信息（需要 Token）
  const profile = await request.get('http://localhost:8080/api/profile', {
    headers: { Authorization: `Bearer ${loginData.token}` }
  });
  expect(profile.status()).toBe(200);
});
```

### 运行测试

```bash
npx playwright test
```
