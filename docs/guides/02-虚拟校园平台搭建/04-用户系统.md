## 4.1 JWT 是什么？

JWT（JSON Web Token）= 一串加密字符串。用户登录后服务器发给他，之后的每次请求带着它来证明"我已登录"。

```
注册: POST /api/auth/register {username, email, password}
  → 返回用户信息

登录: POST /api/auth/login {email, password}
  → 返回 {"token": "eyJhbGci..."}

后续请求: GET /api/profile
  → Header: Authorization: Bearer eyJhbGci...
  → 服务器解析 Token → 获取用户 ID → 返回数据
```

## 4.2 核心代码

### repository/user.go

```go
package repository

import (
    "database/sql"
    "virtual-campus-server/internal/model"
)

type User = model.User

func CreateUser(user *User, password string) error {
    return DB.QueryRow(
        `INSERT INTO users (username, email, password_hash)
         VALUES ($1, $2, $3) RETURNING id, created_at`,
        user.Username, user.Email, password,
    ).Scan(&user.ID, &user.CreatedAt)
}

func GetUserByEmail(email string) (*User, error) {
    u := &User{}
    err := DB.QueryRow(
        `SELECT id, username, email, password_hash, avatar_url, created_at
         FROM users WHERE email = $1`, email,
    ).Scan(&u.ID, &u.Username, &u.Email, &u.PasswordHash, &u.AvatarURL, &u.CreatedAt)
    if err == sql.ErrNoRows { return nil, nil }
    return u, err
}

func GetUserByID(id int) (*User, error) {
    u := &User{}
    err := DB.QueryRow(
        `SELECT id, username, email, avatar_url, created_at
         FROM users WHERE id = $1`, id,
    ).Scan(&u.ID, &u.Username, &u.Email, &u.AvatarURL, &u.CreatedAt)
    if err == sql.ErrNoRows { return nil, nil }
    return u, err
}
```

### service/auth.go

```go
package service

import (
    "errors"
    "time"
    "github.com/golang-jwt/jwt/v5"
    "virtual-campus-server/internal/config"
    "virtual-campus-server/internal/model"
    "virtual-campus-server/internal/repository"
)

type Claims struct {
    UserID   int    `json:"userId"`
    Username string `json:"username"`
    jwt.RegisteredClaims
}

func Register(username, email, password string) (*model.User, error) {
    existing, _ := repository.GetUserByEmail(email)
    if existing != nil { return nil, errors.New("email already registered") }

    user := &model.User{Username: username, Email: email}
    if err := repository.CreateUser(user, password); err != nil {
        return nil, err
    }
    return user, nil
}

func Login(email, password string) (string, *model.User, error) {
    user, _ := repository.GetUserByEmail(email)
    if user == nil || user.PasswordHash != password {
        return "", nil, errors.New("invalid credentials")
    }

    cfg := config.Load()
    claims := &Claims{
        UserID: user.ID, Username: user.Username,
        RegisteredClaims: jwt.RegisteredClaims{
            ExpiresAt: jwt.NewNumericDate(time.Now().Add(24 * time.Hour)),
        },
    }
    token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
    tokenStr, err := token.SignedString([]byte(cfg.JWTSecret))
    return tokenStr, user, err
}
```

### handler/auth.go

```go
package handler

import (
    "net/http"
    "github.com/gin-gonic/gin"
    "virtual-campus-server/internal/repository"
    "virtual-campus-server/internal/service"
)

func Register(c *gin.Context) {
    var req struct {
        Username string `json:"username" binding:"required,min=3,max=50"`
        Email    string `json:"email" binding:"required,email"`
        Password string `json:"password" binding:"required,min=6"`
    }
    if err := c.ShouldBindJSON(&req); err != nil {
        c.JSON(400, gin.H{"error": err.Error()}); return
    }
    user, err := service.Register(req.Username, req.Email, req.Password)
    if err != nil { c.JSON(409, gin.H{"error": err.Error()}); return }
    c.JSON(201, gin.H{"user": user})
}

func Login(c *gin.Context) {
    var req struct {
        Email    string `json:"email" binding:"required,email"`
        Password string `json:"password" binding:"required"`
    }
    if err := c.ShouldBindJSON(&req); err != nil {
        c.JSON(400, gin.H{"error": err.Error()}); return
    }
    token, user, err := service.Login(req.Email, req.Password)
    if err != nil { c.JSON(401, gin.H{"error": err.Error()}); return }
    c.JSON(200, gin.H{"token": token, "user": user})
}

func GetProfile(c *gin.Context) {
    userID, _ := c.Get("userID")
    user, _ := repository.GetUserByID(userID.(int))
    if user == nil { c.JSON(404, gin.H{"error": "not found"}); return }
    c.JSON(200, gin.H{"user": user})
}
```

### middleware/jwt.go

```go
package middleware

import (
    "net/http"
    "strings"
    "github.com/gin-gonic/gin"
    "github.com/golang-jwt/jwt/v5"
    "virtual-campus-server/internal/service"
)

func JWTAuth() gin.HandlerFunc {
    return func(c *gin.Context) {
        h := c.GetHeader("Authorization")
        if h == "" { c.AbortWithStatusJSON(401, gin.H{"error":"missing token"}); return }

        parts := strings.Split(h, " ")
        if len(parts) != 2 || parts[0] != "Bearer" {
            c.AbortWithStatusJSON(401, gin.H{"error":"invalid format"}); return
        }

        claims := &service.Claims{}
        token, err := jwt.ParseWithClaims(parts[1], claims,
            func(t *jwt.Token) (interface{}, error) {
                return []byte("dev-secret-change-in-production"), nil
            })

        if err != nil || !token.Valid {
            c.AbortWithStatusJSON(401, gin.H{"error":"invalid token"}); return
        }

        c.Set("userID", claims.UserID)
        c.Set("username", claims.Username)
        c.Next()
    }
}
```

### 更新 cmd/main.go 路由

```go
api.POST("/auth/register", handler.Register)
api.POST("/auth/login", handler.Login)

auth := api.Group("")
auth.Use(middleware.JWTAuth())
{
    auth.GET("/profile", handler.GetProfile)
}
```

> ⚠️ **安全提醒**：本章用明文存储密码（简化教程）。生产环境必须在 `CreateUser` 中使用 `golang.org/x/crypto/bcrypt` 对密码做哈希。

## 4.3 验证

```bash
# 注册
curl -X POST localhost:8080/api/auth/register \
  -H "Content-Type: application/json" \
  -d '{"username":"test","email":"test@test.com","password":"123456"}'

# 登录
curl -X POST localhost:8080/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"test@test.com","password":"123456"}'
# → {"token":"eyJ...","user":{...}}

# 获取信息（替换 TOKEN）
curl localhost:8080/api/profile -H "Authorization: Bearer TOKEN"
```
